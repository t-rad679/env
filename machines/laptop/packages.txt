# Laptop-only packages. Installed by init/pacman.zsh together with the shared
# list in config/packages.txt (same format: one per line, # for comments, repo
# or AUR).

# Saves/restores the X display layout per monitor set, keyed on EDID, and
# switches profiles automatically on dock/undock via the udev rule + systemd
# units it ships. Profiles live in machines/laptop/autorandr/, linked to
# ~/.config/autorandr by init/links.zsh.
#
# Not cosmetic: without it the laptop boots with the dock's monitors
# "connected but disabled", which makes amdgpu fire a hotplug IRQ every few
# seconds. X re-probes on each one, i3 tears down and rebuilds its bar, and
# every window gets resized twice a second.
autorandr

# Backlight control for the internal panel. Driven via
# machines/laptop/bin/brightness-step from the F7/F8 bindings in i3/laptop.conf.
# Its udev rule grants the *video* group write access to /sys/class/backlight,
# which is why init.zsh below adds the user to that group.
brightnessctl

obs-studio
picard
beets
jetbrains-toolbox

# --- niri / Noctalia / greetd --------------------------------------------
#
# Laptop-only for now: the desktop still runs i3 on X with LightDM, and the
# greetd half of this would take its login screen over. Move these to
# config/packages.txt when that machine follows.

# The compositor. Scrollable tiling rather than i3's tree; config in
# config/niri/config.kdl plus machines/laptop/niri/role.kdl.
niri

# Noctalia Shell: bar, launcher, notifications, control centre, lock screen,
# OSDs and wallpaper in one process, started by `spawn-at-startup` in the niri
# config. It is what replaces i3bar/i3status, dunst, nm-applet, Ulauncher and
# bin/wallpaper on the Wayland side -- all of which stay installed, because
# the i3 session is still selectable from the greeter.
noctalia

# The login screen: greetd owns the VT and runs one greeter;
# noctalia-greeter is a Noctalia-styled one with a session picker. Config in
# machines/laptop/greetd/config.toml, wired up by init.zsh above.
#
# noctalia-greeter is AUR-only (yay handles it); greetd is in extra.
greetd
noctalia-greeter

# X11 apps under niri. niri has no built-in Xwayland -- it integrates
# xwayland-satellite, spawning it on demand the first time something touches
# the X socket and exporting DISPLAY itself. Nothing to configure, but without
# the package every X-only program (JetBrains Toolbox, flameshot, anything
# Electron that refuses Wayland) silently fails to start.
xwayland-satellite

# wl-copy / wl-paste, the Wayland xclip. alias.zsh still assumes xclip, which
# keeps working through xwayland-satellite but only sees the XWayland
# clipboard -- which is not the same selection a native Wayland app uses.
wl-clipboard

# Screen sharing. Noctalia takes its own screenshots, but a portal is what
# Discord, Slack and Meet ask for when they want to share a window, and the
# only backend installed is xdg-desktop-portal-gtk, which implements no
# ScreenCast at all (the same gap that forced flameshot's legacy X11 grab --
# see init/links.zsh). niri speaks org.gnome.Mutter.ScreenCast, so the GNOME
# backend is the one that works with it.
xdg-desktop-portal-gnome

# Qt apps pick X11 without this and come out blurry on the HiDPI panel.
qt6-wayland

# Secret Service provider -- the daemon owning org.freedesktop.secrets, which
# is what Noctalia means by "no Secret Service provider": its calendar
# (Google OAuth refresh token), its own encrypted-storage key and its
# clipboard history all refuse to persist without one. Its error message names
# the three it accepts -- Gnome-keyring, kwallet, keepassXC -- and this is the
# cheap one: no KDE stack, and the only one with a PAM module, so the login
# password unlocks it with no second prompt (see init.zsh).
#
# No spawn-at-startup line to go with it. gnome-keyring ships
# /usr/share/dbus-1/services/org.freedesktop.secrets.service, so the first
# client request D-Bus-activates the daemon inside user@.service.
#
# libsecret -- the client half, which was already installed as a dependency of
# half the desktop -- is not a provider; having it and nothing else is exactly
# the state that produced the error.
#
# Laptop-only for the same reason as the niri block above: it is Noctalia that
# needs it, and auto-unlock is wired into /etc/pam.d/greetd, which only this
# machine has. Chrome also stops falling back to its plaintext `basic`
# password store once a keyring exists, which is the argument for moving this
# to config/packages.txt -- but the desktop would then get an unlock prompt
# that nothing answers, so it moves when lightdm gets the same PAM wiring.
gnome-keyring
