# The user database behind the login screen's avatar. Every greeter that shows
# a face (noctalia-greeter here, lightdm-gtk-greeter on the desktop) reads the
# `IconFile` property off AccountsService; init/avatar.zsh writes it. It was
# already on both machines as a transitive dependency of libmalcontent, which
# is not a dependency to rely on -- hence the explicit entry.
accountsservice

# The xrandr CLI. Spelled xorg-xrandr, not xrandr: there is no package by the
# latter name, so the line that used to be here installed nothing and every
# machine got xrandr only as a dependency of something else.
xorg-xrandr

discord

brightnessctl

# Disk usage analyser -- the GUI for "what filled the SSD this time".
baobab

# Not pulled in by anything here, but assumed by every "extract this" README.
unzip

# Ships /etc/security/limits.d/99-realtime-privileges.conf, which grants the
# `realtime` GROUP rtprio 98, unlimited memlock and nice -11 -- so PipeWire
# can take a real-time scheduling priority instead of falling back and
# xrunning under load.
#
# Installing it is not enough: the drop-in keys on `@realtime`, and neither
# machine's user is in that group yet, so right now this package does
# nothing. `sudo usermod -aG realtime $USER` (then log out) is the other half.
realtime-privileges

# Firmware updates over LVFS (`fwupdmgr refresh && fwupdmgr update`). The
# Framework 16 enumerates as an updatable device, so BIOS/EC updates land
# without building a bootable updater.
fwupd

# Flatpak runtime. Here for OrcaSlicer (com.orcaslicer.OrcaSlicer), which is
# installed as a flatpak rather than from the AUR -- see the `orca` entry in
# machines/laptop/ulauncher/shortcuts.json and the floating rules in
# config/i3/conf.d/orcaslicer.conf. The flatpak itself is not provisioned by
# this repo; `flatpak install flathub com.orcaslicer.OrcaSlicer` is manual.
flatpak

# Interactive screenshot tool: crop, annotate, then copy or save.
# Bound to Print (F11) in config/i3/conf.d/fnrow.conf.
flameshot

# Notification daemon. Owns org.freedesktop.Notifications; without it nothing
# answers notify-send, and flameshot warns "No DBus System Notification service
# found" and drops its "saved as ..." popup on the floor.
#
# config/i3/conf.d/autostart.conf has always exec'd dunst, but the package was
# never in this list, so on a fresh machine that exec failed silently.
dunst

# Color emoji font. Without one, fontconfig has nothing to fall back to for
# emoji codepoints and every emoji renders as a box (tofu): Monarch Money's
# category icons, Discord/Slack reactions, emoji in web pages.
noto-fonts-emoji

# WM-independent dropdown window creator: shows and hides a single window on a
# keypress, Guake-style. Drives the dropdown terminal bound to $mod+grave in
# config/i3/conf.d/baseconfig.conf, which is also where the flags are
# explained. Pulls in xdotool and xorg-xwininfo; its -m (monitor-aware) flag
# needs xorg-xrandr, already above.
tdrop
